# backend/requirements.txt
#
# Dependency manifest for the AI Website Builder's own backend (the builder
# backend, NOT a backend for generated websites — see skills.md 7).
#
# This file exists so the backend layer has a single, reviewable, pinned
# declaration of what it is allowed to import. Every entry traces to an approved
# decision in docs/runtime.md 2. Nothing here is speculative: no second model
# provider, no vector store, no document loader, no ORM, no database driver.
#
# Layer: runtime / configuration.
# Spec:  docs/runtime.md 1-2.1, docs/architecture.md 1 + 3, skills.md 8.
#
# Install target: the repository-root .venv (docs/runtime.md 2).
# Versions are pinned for reproducibility. Resolution has NOT been verified —
# installation is a separate, separately-approved step.

# --- HTTP / API layer -------------------------------------------------------
# FastAPI owns HTTP only; route handlers stay thin and delegate to services.
# (docs/architecture.md 3.2, docs/api.md 1)
fastapi==0.141.1

# ASGI server used to run the FastAPI app. The [standard] extra provides the
# production-grade event loop, HTTP parser and websocket support uvicorn
# recommends. (docs/runtime.md 7)
uvicorn[standard]==0.53.0

# Multipart form parsing. Required by FastAPI for the reference-image upload
# endpoint; without it that route cannot accept a file.
# (docs/api.md 2.4, docs/security.md 4)
python-multipart==0.0.32

# Server-Sent Events for the progress stream. SSE is the documented primary
# progress channel, chosen over WebSocket because the traffic is one-directional.
# (docs/api.md 2.8)
sse-starlette==3.4.11

# --- Schemas & configuration ------------------------------------------------
# Pydantic validates every API request/response, every agent's structured output,
# and the canonical ProjectState. (docs/api.md 4, docs/state.md 2, skills.md 8)
pydantic==2.13.5

# Typed settings loaded from environment variables, with SecretStr redaction and
# fail-fast startup validation. No module reads os.environ directly.
# (docs/runtime.md 3, 6)
pydantic-settings==2.15.0

# --- Workflow orchestration -------------------------------------------------
# LangGraph is the Agent/Workflow Router: graph state, stage transitions,
# conditional routing, HITL interrupt/resume and iteration loops. This is the
# server-side router and has nothing to do with React Router.
# (CLAUDE.md 3A, docs/architecture.md 3.3 + 3.7, docs/state.md 4)
#
# NOTE: the checkpointer package is DEFERRED and deliberately absent — see the
# "Deferred" section at the end of this file.
langgraph==1.2.12

# Durable checkpointer. Replaces the interim InMemorySaver, whose known
# limitation was that a restart lost every pending HITL gate
# (docs/state.md 5.1-5.2). Brings psycopg with it, pinned below.
#
# NOTE: it has no schema option -- its migrations CREATE TABLE unqualified, so
# the tables land wherever search_path points. backend/db/engine.py pins that
# to a dedicated schema; see the long note there for why that is a containment
# boundary and not tidiness.
langgraph-checkpoint-postgres==3.1.2

# --- LLM provider (Anthropic Claude only, v1) -------------------------------
# Prompt composition, runnables and structured-output helpers.
# (docs/architecture.md 3.5, skills.md 8)
langchain-core==1.6.4

# Anthropic Claude binding. Single provider for v1 — one API key serves every
# agent role. No multi-provider abstraction. (CLAUDE.md 7, skills.md 10)
langchain-anthropic==1.7.2

# --- PostgreSQL -------------------------------------------------------------
# psycopg v3, async. Chosen over asyncpg because langgraph-checkpoint-postgres
# speaks psycopg v3, and two drivers for one database is two connection stories
# to reason about. [binary] ships the C extension so no local build toolchain is
# needed; [pool] is the async connection pool backend/db/engine.py builds on.
#
# Deliberately NOT SQLAlchemy or any ORM: the StateRepository Protocol plus
# psycopg is the whole persistence stack, and an ORM would add a second schema
# definition to keep in sync with the Pydantic models (CLAUDE.md 9A).
psycopg[binary,pool]==3.3.6

# --- Outbound HTTP ----------------------------------------------------------
# Used where the backend must make its own HTTP calls. Also the transport the
# above libraries already rely on; pinned here so the version is explicit rather
# than resolved implicitly.
httpx==0.28.1

# --- Rendered-preview capture -----------------------------------------------
# Renders a generated project in headless Chromium so the reviewers granted the
# RENDERED preview have a DOM and screenshots to look at. Without it the
# mandatory quality gate blocks: RenderedPreviewPayload refuses a URL alone, and
# the shipped index.html is an empty <div id="root"> before React runs.
#
# REQUIRES A SEPARATE, NON-pip STEP. The browser is not a Python package:
#
#     python -m playwright install chromium
#
# A fresh checkout that skips it gets a clear launch failure, which the capture
# converts into a BLOCKED gate rather than a silent pass.
playwright==1.63.0

# --- MCP external capability layer ------------------------------------------
# The official SDK, used by backend/tools/mcp_client.py to connect to an MCP
# server and discover its real tools. Chosen over langchain-mcp-adapters, which
# pins mcp<2.0.0 and whose job -- handing MCP tools straight to an agent -- is
# the thing docs/mcp.md 3 requires us NOT to do.
#
# Optional at runtime: docs/mcp.md 1 requires the core workflow to run with zero
# MCP servers connected, and every SDK import is local to the connect path, so
# the graph imports without it.
mcp==2.3.0

# --- Testing ----------------------------------------------------------------
# Runs the verification suites in backend/tests. Not needed to serve the API, but
# pinned with everything else so a checkout tests identically.
#
# pytest-asyncio is deliberately NOT here: the suites drive async code with
# asyncio.run() inside standalone scripts, so no event-loop plugin is involved.
# Add it only if a suite is ever written as an async test function.
pytest==9.1.1


# ============================================================================
# DEFERRED — do NOT add without an explicit decision (docs/runtime.md 2.1)
# ============================================================================
#
#   Supabase client library (@supabase / supabase-py)
#       Status:      DEFERRED, and may never be needed.
#       Why:         The backend reaches PostgreSQL directly with psycopg. A
#                    Supabase client exists to call PostgREST, Storage and the
#                    Auth admin API -- none of which v1 uses. Supabase Auth is
#                    verified by JWT, not by an SDK (docs/database.md 1.3.5).
#       Unblocked by: a requirement that actually needs one of those three.
#
#   JWT verification library (PyJWT[crypto] / joserfc)
#       Status:      REQUIRED for authentication, NOT YET ADDED.
#       Why:         The Supabase project signs access tokens with ES256 and
#                    publishes a JWKS, so verification needs an asymmetric-capable
#                    library. There is no shared secret to compare against, and
#                    SUPABASE_JWT_SECRET must not be introduced.
#       Unblocked by: the auth implementation step.
#
#   Migration tool (Alembic or similar)
#       Status:      NOT CHOSEN. Plain SQL files under backend/db/migrations/
#                    are the current approach (docs/runtime.md 2.1).
#       Why:         Alembic is ORM-adjacent and the project has no ORM.
#
#   ORM / SQLAlchemy
#       Status:      NOT PERMITTED. See the psycopg entry above.
#
#   Impeccable
#       Status:      REQUIRED, and INSTALLED -- Claude Code skill 4.3.1 with CLI
#                    4.0.0. Not a pip dependency; it is a separate tool, which is
#                    why it is listed here rather than above.
#       Note:        This entry previously claimed Impeccable was not installed.
#                    That was wrong, and it mattered: code written to that premise
#                    would report the mandatory gate blocked forever on a machine
#                    where the tool works (docs/design.md 5.2).
#
#   Zapier client / webhook tooling
#       Status:      DEFERRED. Optional automation layer only. Not the database,
#                    not the orchestrator. (CLAUDE.md 19A, docs/mcp.md 2.4)
#
#   RAG stack (vector store, embeddings, document loaders)
#       Status:      DEFERRED. Nothing in v1 scope requires retrieval.
#                    (CLAUDE.md 22, skills.md 24)
#
#   Second model provider
#       Status:      NOT PERMITTED in v1. (CLAUDE.md 7, skills.md 10)
#
#   Test / lint tooling
#       Status:      Not configured for this project. Adding any would be a
#                    separate, explicit decision. (docs/workflow.md 7.1)
